# RoadVision ML / CV dependencies.
#
# Kept separate from requirements.txt so the API can be installed and started
# quickly without pulling in torch (a large download).
#
#   pip install -r requirements-ml.txt
#
# Ranges rather than exact pins on purpose: ultralytics must be free to pull a
# torch/numpy combination that has wheels for whichever Python version is on
# the machine. Over-pinning numpy is the usual cause of a failed install right
# before a demo, and opencv is left to ultralytics so only one copy installs.

# YOLOv8 detector. Brings torch, torchvision, opencv and numpy with it.
ultralytics>=8.3,<9

# Used by app/services/ml_service.py to read image dimensions.
pillow>=10.0

# Random Forest priority classifier (app/ml/priority/predictor.py).
# Train and load the bundle with the SAME scikit-learn major.minor version,
# otherwise joblib will warn or fail on unpickling.
scikit-learn>=1.3
joblib>=1.3

# CPU-only machines: run this FIRST for a much smaller install (no CUDA),
# then install the requirements above.
#   pip install torch torchvision --index-url https://download.pytorch.org/whl/cpu
